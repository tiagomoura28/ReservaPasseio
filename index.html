/** * Sertão Tour — Google Sheets + InfinitePay Checkout Integrado * * Propriedades do script (Configurações do projeto > Propriedades do script): * SPREADSHEET_ID = ID da planilha privada * INFINITEPAY_HANDLE = InfiniteTag sem $ * INFINITEPAY_API_KEY = API Key do Checkout Integrado (inchk_...) * REDIRECT_URL = URL HTTPS da página de retorno * * Implante como Aplicativo da Web, executar como você, acesso conforme necessário. * Nunca coloque a API Key no HTML. */ const SHEET_NAME = 'Reservas'; const CHECKOUT_ENDPOINT = 'https://api.checkout.infinitepay.io/links'; function doPost(e) { try { const params = (e && e.parameter) || {}; // Fluxo do navegador: responde com uma página que redireciona ao checkout. // Evita depender de fetch/CORS do navegador para ler JSON do Apps Script. if (params.browserCheckout === 'true') { const booking = JSON.parse(params.bookingJson || '{}'); const result = createReservation_(booking); return redirectToCheckout_(result.checkoutUrl); } const body = JSON.parse((e && e.postData && e.postData.contents) || '{}'); if (body.action === 'createReservation') { return json_(createReservation_(body.booking || {})); } if (body.action === 'paymentWebhook') { return json_(processPaymentWebhook_(body)); } return json_({success:false, message:'Ação não reconhecida.'}); } catch (err) { console.error(err && err.stack ? err.stack : String(err)); // O fluxo via formulário exibe uma página legível em caso de erro. if (e && e.parameter && e.parameter.browserCheckout === 'true') { return errorPage_(String(err && err.message ? err.message : err)); } return json_({success:false, message:String(err && err.message ? err.message : err)}); } } function createReservation_(b) { validateBooking_(b); const props = PropertiesService.getScriptProperties(); const apiKey = props.getProperty('INFINITEPAY_API_KEY'); const handle = props.getProperty('INFINITEPAY_HANDLE'); const redirectUrl = props.getProperty('REDIRECT_URL'); const spreadsheetId = props.getProperty('SPREADSHEET_ID'); if (!apiKey || !handle || !redirectUrl || !spreadsheetId) { throw new Error('Configure SPREADSHEET_ID, INFINITEPAY_HANDLE, INFINITEPAY_API_KEY e REDIRECT_URL nas propriedades do script.'); } const sheet = getSheet_(spreadsheetId); ensureHeaders_(sheet); const orderNsu = Utilities.getUuid(); const rowNumber = sheet.getLastRow() + 1; // Registra primeiro a reserva como pendente, conforme solicitado. const row = [ new Date(), orderNsu, b.passeio || 'Passeio ao Paraíso — Marina Hotel', b.nome, b.nascimento, b.telefone, b.email || '', b.dataPasseio, Number(b.adultos), Number(b.criancas05Qtd || 0), b.criancaPequena || '', b.nascPequena || '', Number(b.criancas612Qtd || 0), b.criancaMaior || '', b.nascMaior || '', Number(b.valorTotal), 'Pendente', '', '' ]; sheet.getRange(rowNumber, 1, 1, row.length).setValues([row]); SpreadsheetApp.flush(); const items = []; const adults = Number(b.adultos); const kids612 = Number(b.criancas612Qtd || 0); if (adults > 0) items.push({quantity:adults, price:18900, description:'Passeio ao Paraíso — adulto'}); if (kids612 > 0) items.push({quantity:kids612, price:16400, description:'Passeio ao Paraíso — criança de 6 a 12 anos'}); if (!items.length) throw new Error('O pedido não contém itens tarifados.'); // A documentação atual do Checkout Integrado usa handle, order_nsu, // redirect_url, items e customer; a API Key é enviada no cabeçalho. const redirect = redirectUrl + (redirectUrl.indexOf('?') >= 0 ? '&' : '?') + 'order_nsu=' + encodeURIComponent(orderNsu); const payload = { handle: handle, order_nsu: orderNsu, redirect_url: redirect, items: items, customer: { name: String(b.nome || ''), email: String(b.email || ''), phone_number: normalizePhone_(b.telefone) }, options: { single_payment: true, payment_methods_enabled: ['credit_card', 'pix'] } }; const response = UrlFetchApp.fetch(CHECKOUT_ENDPOINT, { method:'post', contentType:'application/json', headers:{'X-Api-Key':apiKey}, payload:JSON.stringify(payload), muteHttpExceptions:true }); const code = response.getResponseCode(); const responseText = response.getContentText(); let data; try { data = JSON.parse(responseText || '{}'); } catch (_) { data = {}; } if (code < 200 || code >= 300) { sheet.getRange(rowNumber, 17).setValue('Erro ao criar checkout (HTTP ' + code + ')'); console.error('InfinitePay HTTP ' + code + ': ' + responseText); throw new Error('A InfinitePay recusou a criação do checkout (HTTP ' + code + '). Verifique a API Key, a InfiniteTag e a habilitação do Checkout Integrado. A reserva ficou registrada como erro.'); } const checkoutUrl = data.url || data.checkout_url || data.link; if (!checkoutUrl || !/^https:\/\/(checkout\.infinitepay\.com\.br|pay\.infinitepay\.io|checkout\.infinitepay\.io)\//i.test(checkoutUrl)) { sheet.getRange(rowNumber, 17).setValue('Erro: URL de checkout ausente'); console.error('Resposta inesperada da InfinitePay: ' + responseText); throw new Error('A InfinitePay respondeu, mas não retornou uma URL de checkout válida. Consulte o registro de execução do Apps Script.'); } sheet.getRange(rowNumber, 18).setValue(checkoutUrl); return {success:true, orderNsu:orderNsu, checkoutUrl:checkoutUrl}; } function validateBooking_(b) { if (!b || !b.nome || !b.nascimento || !b.telefone || !b.dataPasseio) { throw new Error('Preencha nome, nascimento, telefone e data desejada.'); } const adults = Number(b.adultos); const kids612 = Number(b.criancas612Qtd || 0); const kids05 = Number(b.criancas05Qtd || 0); if (!Number.isInteger(adults) || adults < 1 || adults > 10) throw new Error('Quantidade de adultos inválida.'); if (![0,1].includes(kids612) || ![0,1].includes(kids05)) throw new Error('Quantidade de crianças inválida.'); const expected = adults * 18900 + kids612 * 16400; if (Number(b.valorTotal) !== expected) throw new Error('O valor informado não corresponde à tabela de preços.'); } function processPaymentWebhook_(event) { // Este método NÃO confia em chamadas arbitrárias do navegador para marcar pago. // Configure o webhook na InfinitePay e valide a autenticidade/evento conforme // os recursos habilitados para sua conta antes de usar esta rotina em produção. const orderNsu = String(event.order_nsu || event.orderNsu || ''); if (!orderNsu) throw new Error('Webhook sem order_nsu.'); const spreadsheetId = PropertiesService.getScriptProperties().getProperty('SPREADSHEET_ID'); if (!spreadsheetId) throw new Error('SPREADSHEET_ID não configurado.'); const sheet = getSheet_(spreadsheetId); ensureHeaders_(sheet); const lastRow = sheet.getLastRow(); if (lastRow < 2) throw new Error('Nenhuma reserva encontrada.'); const ids = sheet.getRange(2, 2, lastRow - 1, 1).getDisplayValues(); const index = ids.findIndex(r => r[0] === orderNsu); if (index < 0) throw new Error('Pedido não encontrado.'); const row = index + 2; const status = String(event.event || event.status || '').toLowerCase(); if (status === 'paid' || status === 'settled' || event.paid === true) { sheet.getRange(row, 17).setValue('Pago'); } else if (status === 'refunded') { sheet.getRange(row, 17).setValue('Estornado'); } else if (status === 'canceled' || status === 'payment_failed') { sheet.getRange(row, 17).setValue(status === 'canceled' ? 'Cancelado' : 'Falha no pagamento'); } if (event.transaction_nsu) sheet.getRange(row, 19).setValue(String(event.transaction_nsu)); return {success:true}; } function getSheet_(spreadsheetId) { const ss = SpreadsheetApp.openById(spreadsheetId); return ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME); } function ensureHeaders_(sheet) { if (sheet.getLastRow() === 0) { sheet.appendRow([ 'Criada em','ID do pedido','Passeio','Responsável','Nascimento responsável','Telefone', 'E-mail','Data desejada','Adultos','Qtd 0-5','Nome 0-5','Nascimento 0-5', 'Qtd 6-12','Nome 6-12','Nascimento 6-12','Total (centavos)','Status pagamento', 'URL checkout','ID transação' ]); sheet.setFrozenRows(1); } } function normalizePhone_(phone) { const digits = String(phone || '').replace(/\D/g, ''); return digits ? (digits.startsWith('55') ? '+' + digits : '+55' + digits) : ''; } function redirectToCheckout_(url) { const safeUrl = JSON.stringify(url); return HtmlService.createHtmlOutput( '' + 'Redirecionando para pagamento' + '

Reserva registrada. Redirecionando para o checkout seguro da InfinitePay…

' + '

Se não for redirecionado, clique aqui para pagar.

' + '' ); } function errorPage_(message) { const safe = String(message).replace(/&/g,'&').replace(//g,'>'); return HtmlService.createHtmlOutput( '' + '' + '

Não foi possível abrir o pagamento

' + safe + '

' + '

A reserva pode ter sido registrada. Entre em contato com a agência antes de tentar novamente para evitar duplicidade.

' + '' ); } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
